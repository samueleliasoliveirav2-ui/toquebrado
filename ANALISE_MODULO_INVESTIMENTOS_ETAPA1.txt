🗃️ 3. MODELOS / TABELAS NECESSÁRIAS (4 tabelas + 2 enums PG)
================================================================================
--- Modelo 0: ENUMs PostgreSQL ---
CREATE TYPE public.tipo_renda_investimento AS ENUM (
  'RENDA_FIXA','RENDA_VARIAVEL','FUNDOS','CRIPTO','POUPANCA','PREVIDENCIA','OUTROS'
);
CREATE TYPE public.status_investimento AS ENUM (
  'EM_CURSO','RESGATADO','VENCIDO','EM_LIQUIDACAO'
);

--- Modelo 1: public.investimentos (cadastro mestre 1 linha = 1 investimento) ---
Campos:
- id UUID PK gen_random_uuid()
- userId UUID FK auth.users(id) NOT NULL (RLS)
- nome VARCHAR(120) NOT NULL (apelido usuario: "CDB Nubank 2026", "FII HGLG")
- tipoRenda tipo_renda_investimento NOT NULL
- subCategoria VARCHAR(80) NOT NULL ("CDB","Ação","FII","ETF","BDR","BTC","ETH")
- instituicao VARCHAR(120) NULL (Nubank, XP, Binance)
- dataCompra DATE NOT NULL
- dataVencimento DATE NULL (opcional: RF geral tem, Ação não)
- valorAportadoInicial NUMERIC(18,4) NOT NULL
- quantidadeCotas NUMERIC(24,10) NULL (Ação/FIIs/Fundos/Cripto)
- precoCotaCompra NUMERIC(24,10) NULL
- valorAtualEstimado NUMERIC(18,4) NOT NULL (atualiza via históricos)
- taxaAno VARCHAR(80) NULL ("120% CDI", "IPCA +6%", "14,5% a.a.")
- liquidez VARCHAR(60) NOT NULL ("Diária","Vencimento","D+1","D+30")
- status status_investimento NOT NULL DEFAULT 'EM_CURSO'
- observacoes TEXT NULL
- carteira VARCHAR(60) NULL ("Reserva Emergência","Aposentadoria")
- created_at / updated_at timestamptz (trigger)
Índices: (userId, status, tipoRenda, dataVencimento)

--- Modelo 2: public.investimento_movimentacoes (aportes / retiradas) ---
- id / userId / investimentoId FK investimentos(id) ON DELETE CASCADE
- tipo ENUM('APORTE','RETIRADA') NOT NULL
- dataMovimento DATE NOT NULL
- valor NUMERIC(18,4) NOT NULL
- quantidadeCotasMovimentadas / precoCotaMovimento (numérico opcional)
- observacoes TEXT NULL, created_at timestamptz
Índices: (userId, investimentoId, dataMovimento); (userId, tipo, dataMovimento)

--- Modelo 3: public.investimento_historico_valores (histórico valorização) ---
- id / userId / investimentoId FK
- dataReferencia DATE NOT NULL
- valorPatrimonioLiquido NUMERIC(18,4) NOT NULL
- quantidadeCotasAtualizada / precoCotaAtual (numérico opcional)
- fonteDado ENUM('MANUAL','CALCULADO','IMPORTADO_CSV') NOT NULL DEFAULT 'MANUAL'
- observacoes / created_at
UNIQUE(userId, investimentoId, dataReferencia) (1 registro por investimento + dia)

--- Modelo 4: public.investimento_categorias_custom (plano contas usuario) ---
- id / userId
- nome VARCHAR(80) NOT NULL ("Ações BR", "Cripto Lunar")
- tipoRenda tipo_renda_investimento NOT NULL
- ehPadraoSistema BOOL NOT NULL default false
- created_at timestamptz

🔄 6. FUNCIONAMENTO FUTURO (APORTES / RETIRADAS / HISTÓRICO)
================================================================================
6.1 Registrar Aporte:
  INSERT investimento_movimentacoes tipo=APORTE
  => atualiza investimentos.valorAtualEstimado += valor
  => INSERT automático investimento_historico_valores fonte=CALCULADO

6.2 Registrar Retirada / Resgate:
  INSERT movimentacoes tipo=RETIRADA
  => valorAtualEstimado -= valor; se = 0 auto status='RESGATADO'
  => histórico CALCULADO

6.3 Atualizar valor manual (1x/mês usu informar):
  INSERT histórico fonte=MANUAL
  => atualiza valorAtualEstimado
  => Hero investimentos calcula rentab % (atual vs total aportado)
